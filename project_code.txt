package ru.campuseats.config;

import io.swagger.v3.oas.models.OpenAPI;
import io.swagger.v3.oas.models.info.Contact;
import io.swagger.v3.oas.models.info.Info;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.boot.CommandLineRunner;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.core.io.ClassPathResource;
import org.springframework.jdbc.datasource.init.ResourceDatabasePopulator;
import org.springframework.web.servlet.config.annotation.CorsRegistry;
import org.springframework.web.servlet.config.annotation.ResourceHandlerRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;
import ru.campuseats.repository.UserRepository;
import javax.sql.DataSource;

@Slf4j
@Configuration
@RequiredArgsConstructor
public class AppConfig implements WebMvcConfigurer {

    private final UserRepository userRepository;
    private final DataSource dataSource;

    @Bean
    public OpenAPI campusEatsOpenAPI() {
        return new OpenAPI()
                .info(new Info()
                        .title("CampusEats REST API")
                        .version("1.0.0")
                        .description("Сервис предзаказа еды в точках питания кампуса без очередей")
                        .contact(new Contact()
                                .name("CampusEats Team")
                                .email("admin@campus.edu")));
    }

    @Override
    public void addCorsMappings(CorsRegistry registry) {
        registry.addMapping("/api/**")
                .allowedOrigins("*")
                .allowedMethods("GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS")
                .allowedHeaders("*");
    }

    @Override
    public void addResourceHandlers(ResourceHandlerRegistry registry) {
        registry.addResourceHandler("/uploads/**")
                .addResourceLocations("file:./uploads/");
    }

    @Bean
    public CommandLineRunner initDatabaseFromSql() {
        return args -> {
            if (userRepository.count() == 0) {
                log.info("База данных пуста. Выполнение SQL-скрипта инициализации classpath:data.sql...");
                ResourceDatabasePopulator populator = new ResourceDatabasePopulator(
                        false, false, "UTF-8", new ClassPathResource("data.sql")
                );
                populator.execute(dataSource);
                log.info("SQL-скрипт data.sql успешно выполнен (заполнены все 13 таблиц).");
            } else {
                log.info("База данных уже содержит данные — пропуск выполнения data.sql.");
            }
        };
    }
}

    @GetMapping({"", "/", "/ui"})
    public String showMainPage(
            @RequestParam(defaultValue = "default") String tab,
            @RequestParam(defaultValue = "login") String authMode,
            @RequestParam(required = false) String search,
            @RequestParam(required = false) Long canteenId,
            @RequestParam(required = false) Long categoryId,
            @RequestParam(required = false) String dietaryTagCode,
            @RequestParam(required = false) BigDecimal maxPrice,
            @RequestParam(required = false) Integer maxCalories,
            @RequestParam(defaultValue = "rating") String sortBy,
            @RequestParam(required = false) Long selectedDishId,
            @RequestParam(required = false) Long editingDishId,
            @RequestParam(required = false) Long editingCanteenId,
            @RequestParam(required = false) Long trackingOrderId,
            HttpSession session,
            Model model
    ) {
        Long userId = (Long) session.getAttribute(SESSION_USER_ID);

        if (userId == null) {
            model.addAttribute("currentUser", null);
            model.addAttribute("authMode", authMode);
            return "index";
        }

        UserDto currentUser;
        try {
            currentUser = userService.getUserById(userId);
        } catch (Exception e) {
            session.invalidate();
            return "redirect:/";
        }

        model.addAttribute("currentUser", currentUser);
        model.addAttribute("categories", dishCatalogService.getAllCategories());
        model.addAttribute("dietaryTags", dishCatalogService.getAllDietaryTags());

        // Уведомления нужны студенту (статусы заказов) и точке питания (новые заказы и верификация)
        if (currentUser.getRole() != Role.ADMIN) {
            model.addAttribute("notifications", notificationService.getUserNotifications(userId));
            model.addAttribute("unreadCount", notificationService.getUnreadCount(userId));
        }

        // ---------------- РОЛЬ 1: СТУДЕНТ ----------------
        if (currentUser.getRole() == Role.STUDENT) {
            model.addAttribute("canteens", canteenService.getVerifiedCanteens());
            String studentTab = "default".equals(tab) ? "catalog" : tab;
            model.addAttribute("activeTab", studentTab);
            model.addAttribute("cart", cartService.getUserCart(userId));

            if ("catalog".equals(studentTab)) {
                model.addAttribute("search", search);
                model.addAttribute("canteenId", canteenId);
                model.addAttribute("categoryId", categoryId);
                model.addAttribute("dietaryTagCode", dietaryTagCode);
                model.addAttribute("maxPrice", maxPrice);
                model.addAttribute("maxCalories", maxCalories);
                model.addAttribute("sortBy", sortBy);

                model.addAttribute("dishes", dishCatalogService.searchAndFilterDishes(
                        search, canteenId, categoryId, dietaryTagCode, maxPrice, maxCalories, true, sortBy
                ));
            }

            if ("dish".equals(studentTab) && selectedDishId != null) {
                DishDetailDto detail = dishCatalogService.getDishDetails(selectedDishId);
                if (!Boolean.TRUE.equals(detail.getIsAvailable())) {
                    return "redirect:/?tab=catalog";
                }
                model.addAttribute("selectedDish", detail);
                model.addAttribute("dishReviews", reviewService.getReviewsForDish(selectedDishId));
            }

            if ("favorites".equals(studentTab)) {
                model.addAttribute("favorites", favoriteService.getUserFavorites(userId));
            }

            if ("orders".equals(studentTab)) {
                List<OrderResponseDto> orders = orderService.getUserOrderHistory(userId, false);
                model.addAttribute("studentOrders", orders);

                Long orderToTrack = (trackingOrderId != null)
                        ? trackingOrderId
                        : (orders.isEmpty() ? null : orders.get(0).getId());

                if (orderToTrack != null) {
                    model.addAttribute("tracking", orderService.trackOrder(orderToTrack));
                }
            }
        }

        // ---------------- РОЛЬ 2: ТОЧКА ПИТАНИЯ (VENDOR) ----------------
        if (currentUser.getRole() == Role.VENDOR) {
            String vendorTab = "default".equals(tab) ? "vendor_orders" : tab;
            model.addAttribute("activeTab", vendorTab);

            CanteenDto myCanteen = canteenService.getCanteenByManagerId(userId);
            Long myCanteenId = myCanteen.getId();
            model.addAttribute("vendorCanteen", myCanteen);
            model.addAttribute("vendorCanteenId", myCanteenId);

            if ("vendor_orders".equals(vendorTab)) {
                model.addAttribute("vendorOrders", vendorOrderService.getCanteenQueue(myCanteenId, null));
            }
            if ("vendor_menu".equals(vendorTab)) {
                model.addAttribute("vendorDishes", dishCatalogService.searchAndFilterDishes(
                        null, myCanteenId, null, null, null, null, false, "rating"
                ));
                if (editingDishId != null) {
                    model.addAttribute("editingDish", dishCatalogService.getDishDetails(editingDishId));
                }
            }
            if ("vendor_stats".equals(vendorTab)) {
                model.addAttribute("vendorAnalytics", analyticsService.getCampusDashboard(myCanteenId));
            }
            if ("vendor_reviews".equals(vendorTab)) {
                model.addAttribute("vendorReviews", reviewService.getReviewsForCanteen(myCanteenId));
            }
        }

        // ---------------- РОЛЬ 3: АДМИНИСТРАТОР (ADMIN) ----------------
        if (currentUser.getRole() == Role.ADMIN) {
            String adminTab = "default".equals(tab) ? "admin_dashboard" : tab;
            model.addAttribute("activeTab", adminTab);

            if ("admin_dashboard".equals(adminTab)) {
                model.addAttribute("canteens", canteenService.getAllCanteens(false));
                model.addAttribute("analytics", analyticsService.getCampusDashboard(null));
                if (editingCanteenId != null) {
                    model.addAttribute("editingCanteen", canteenService.getCanteenById(editingCanteenId));
                }
            }
        }

        return "index";
    }

package ru.campuseats.repository;

import jakarta.persistence.criteria.Join;
import jakarta.persistence.criteria.JoinType;
import org.springframework.data.jpa.domain.Specification;
import ru.campuseats.entity.DietaryTag;
import ru.campuseats.entity.Dish;

import java.math.BigDecimal;
import java.util.Objects;
import java.util.stream.Stream;

public final class DishSpecification {

    private DishSpecification() {
    }

    public static Specification<Dish> filterDishes(
            String search,
            Long canteenId,
            Long categoryId,
            String dietaryTagCode,
            BigDecimal maxPrice,
            Integer maxCalories,
            Boolean onlyAvailable
    ) {
        return Stream.of(
                        bySearch(search),
                        byCanteen(canteenId),
                        byCategory(categoryId),
                        byDietaryTag(dietaryTagCode),
                        byMaxPrice(maxPrice),
                        byMaxCalories(maxCalories),
                        byOnlyAvailable(onlyAvailable)
                )
                .filter(Objects::nonNull)
                .reduce(Specification.where(null), Specification::and);
    }

    private static Specification<Dish> bySearch(String search) {
        if (search == null || search.isBlank()) return null;
        String pattern = "%" + search.trim().toLowerCase() + "%";
        return (root, query, cb) -> cb.or(
                cb.like(cb.lower(root.get("name")), pattern),
                cb.like(cb.lower(root.get("composition")), pattern)
        );
    }

    private static Specification<Dish> byCanteen(Long canteenId) {
        return canteenId == null ? null :
                (root, query, cb) -> cb.equal(root.get("canteen").get("id"), canteenId);
    }

    private static Specification<Dish> byCategory(Long categoryId) {
        return categoryId == null ? null :
                (root, query, cb) -> cb.equal(root.get("category").get("id"), categoryId);
    }

    private static Specification<Dish> byDietaryTag(String tagCode) {
        if (tagCode == null || tagCode.isBlank()) return null;
        return (root, query, cb) -> {
            query.distinct(true);
            Join<Dish, DietaryTag> tagJoin = root.join("dietaryTags", JoinType.INNER);
            return cb.equal(cb.upper(tagJoin.get("code")), tagCode.trim().toUpperCase());
        };
    }

    private static Specification<Dish> byMaxPrice(BigDecimal maxPrice) {
        return maxPrice == null ? null :
                (root, query, cb) -> cb.lessThanOrEqualTo(root.get("price"), maxPrice);
    }

    private static Specification<Dish> byMaxCalories(Integer maxCalories) {
        return maxCalories == null ? null :
                (root, query, cb) -> cb.lessThanOrEqualTo(root.get("calories"), maxCalories);
    }

    private static Specification<Dish> byOnlyAvailable(Boolean onlyAvailable) {
        return !Boolean.TRUE.equals(onlyAvailable) ? null :
                (root, query, cb) -> cb.and(
                        cb.isTrue(root.get("isAvailable")),
                        cb.isTrue(root.get("canteen").get("isVerified")),
                        cb.isTrue(root.get("canteen").get("isOpen"))
                );
    }
}

package ru.campuseats.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import ru.campuseats.dto.CampusAnalyticsDto;
import ru.campuseats.dto.CanteenWorkloadDto;
import ru.campuseats.dto.PopularDishStatDto;
import ru.campuseats.entity.*;
import ru.campuseats.enums.OrderStatus;
import ru.campuseats.mapper.AnalyticsMapper;
import ru.campuseats.repository.*;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.*;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class AnalyticsService {

    private final FoodOrderRepository foodOrderRepository;
    private final CanteenRepository canteenRepository;
    private final DishRepository dishRepository;
    private final UserRepository userRepository;
    private final AnalyticsMapper analyticsMapper;

    @Transactional(readOnly = true)
    public CampusAnalyticsDto getCampusDashboard(Long canteenIdFilter) {
        List<FoodOrder> allOrders = (canteenIdFilter != null)
                ? foodOrderRepository.findByCanteenIdOrderByCreatedAtDesc(canteenIdFilter)
                : foodOrderRepository.findAll();

        int totalOrders = allOrders.size();
        int activeOrders = (int) allOrders.stream()
                .filter(o -> o.getStatus() == OrderStatus.ACCEPTED
                        || o.getStatus() == OrderStatus.COOKING
                        || o.getStatus() == OrderStatus.READY)
                .count();
        int completedOrders = (int) allOrders.stream()
                .filter(o -> o.getStatus() == OrderStatus.COMPLETED)
                .count();

        List<FoodOrder> validOrders = allOrders.stream()
                .filter(o -> o.getStatus() != OrderStatus.CANCELLED)
                .toList();

        BigDecimal totalRevenue = validOrders.stream()
                .map(FoodOrder::getTotalAmount)
                .reduce(BigDecimal.ZERO, BigDecimal::add)
                .setScale(2, RoundingMode.HALF_UP);

        BigDecimal averageCheck = validOrders.isEmpty()
                ? BigDecimal.ZERO
                : totalRevenue.divide(BigDecimal.valueOf(validOrders.size()), 2, RoundingMode.HALF_UP);

        Map<String, Long> ordersByStatus = new LinkedHashMap<>();
        for (OrderStatus status : OrderStatus.values()) {
            long count = allOrders.stream().filter(o -> o.getStatus() == status).count();
            ordersByStatus.put(status.name(), count);
        }

        Map<String, Long> ordersByTimeSlot = allOrders.stream()
                .collect(Collectors.groupingBy(
                        FoodOrder::getPickupTimeSlot,
                        LinkedHashMap::new,
                        Collectors.counting()
                ));

        List<Canteen> canteens = (canteenIdFilter != null)
                ? canteenRepository.findById(canteenIdFilter).map(List::of).orElse(Collections.emptyList())
                : canteenRepository.findAll();

        List<CanteenWorkloadDto> workloads = canteens.stream()
                .map(this::buildCanteenWorkload)
                .toList();

        List<PopularDishStatDto> topDishes = calculateTopDishes(validOrders, 5);

        return analyticsMapper.toCampusAnalyticsDto(
                totalOrders,
                activeOrders,
                completedOrders,
                totalRevenue,
                averageCheck,
                userRepository.count(),
                dishRepository.count(),
                ordersByStatus,
                ordersByTimeSlot,
                workloads,
                topDishes
        );
    }

    private CanteenWorkloadDto buildCanteenWorkload(Canteen canteen) {
        List<FoodOrder> canteenOrders = foodOrderRepository.findByCanteenIdOrderByCreatedAtDesc(canteen.getId());
        int totalCount = canteenOrders.size();

        int activeQueue = (int) canteenOrders.stream()
                .filter(o -> o.getStatus() == OrderStatus.ACCEPTED
                        || o.getStatus() == OrderStatus.COOKING
                        || o.getStatus() == OrderStatus.READY)
                .count();

        int slotCapacity = 5;
        int workloadPercent = Math.min(100, Math.round((activeQueue * 100.0f) / slotCapacity));

        String workloadLevel;
        if (workloadPercent >= 75) {
            workloadLevel = "Час пик";
        } else if (workloadPercent >= 40) {
            workloadLevel = "Средняя";
        } else {
            workloadLevel = "Свободно";
        }

        BigDecimal revenue = canteenOrders.stream()
                .filter(o -> o.getStatus() != OrderStatus.CANCELLED)
                .map(FoodOrder::getTotalAmount)
                .reduce(BigDecimal.ZERO, BigDecimal::add)
                .setScale(2, RoundingMode.HALF_UP);

        return analyticsMapper.toCanteenWorkloadDto(
                canteen,
                totalCount,
                activeQueue,
                slotCapacity,
                workloadPercent,
                workloadLevel,
                revenue
        );
    }

    private List<PopularDishStatDto> calculateTopDishes(List<FoodOrder> validOrders, int limit) {
        Map<Long, Dish> dishById = new HashMap<>();
        Map<Long, Integer> portionsByDish = new HashMap<>();
        Map<Long, BigDecimal> revenueByDish = new HashMap<>();

        for (FoodOrder order : validOrders) {
            for (OrderItem item : order.getItems()) {
                Dish dish = item.getDish();
                if (dish == null) continue;
                Long dishId = dish.getId();
                dishById.put(dishId, dish);
                portionsByDish.merge(dishId, item.getQuantity(), Integer::sum);
                BigDecimal itemRevenue = item.getUnitPrice().multiply(BigDecimal.valueOf(item.getQuantity()));
                revenueByDish.merge(dishId, itemRevenue, BigDecimal::add);
            }
        }

        return portionsByDish.entrySet().stream()
                .sorted(Map.Entry.<Long, Integer>comparingByValue().reversed())
                .limit(limit)
                .map(entry -> {
                    Long dishId = entry.getKey();
                    return analyticsMapper.toPopularDishStatDto(
                            dishById.get(dishId),
                            entry.getValue(),
                            revenueByDish.getOrDefault(dishId, BigDecimal.ZERO).setScale(2, RoundingMode.HALF_UP)
                    );
                })
                .toList();
    }
}

                    ? "Администратор кампуса подтвердил официальный статус точки «" + saved.getName() + "». Теперь ваше меню доступно студентам!"

            dishCatalogService.deleteDish(dish.getId());

        List<FoodOrder> orders = foodOrderRepository.findByCanteenIdOrderByCreatedAtDesc(canteenId);

        }

        canteenRepository.delete(canteen);
    }

@Service

    private final CartItemRepository cartItemRepository;
    private final UserRepository userRepository;
    private final DishRepository dishRepository;
    private final CartMapper cartMapper;

    private static final List<String> DEFAULT_PICKUP_SLOTS = List.of(
            "10:15 - 10:30 (Перемена)",
            "11:50 - 12:05 (Большая перемена)",
            "12:15 - 12:30 (Большая перемена)",

    public CartSummaryDto getUserCart(Long userId) {

        return buildCartSummary(userId, items);
    }

    @Transactional
    public CartSummaryDto addItemToCart(AddToCartRequestDto request) {
        User user = findUserOrThrow(request.getUserId());
        Dish dish = dishRepository.findById(request.getDishId())
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND, "Блюдо с ID " + request.getDishId() + " не найдено"
                ));

        if (!Boolean.TRUE.equals(dish.getIsAvailable())) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST, "Блюдо «" + dish.getName() + "» закончилось и недоступно для заказа"
            );
        }

        List<CartItem> currentItems = cartItemRepository.findByUserIdOrderByAddedAtAsc(user.getId());

        if (!currentItems.isEmpty()) {
            Canteen existingCanteen = currentItems.get(0).getDish().getCanteen();
            if (!existingCanteen.getId().equals(dish.getCanteen().getId())) {
                if (Boolean.TRUE.equals(request.getClearIfDifferentCanteen())) {
                    cartItemRepository.deleteByUserId(user.getId());
                    cartItemRepository.flush();
                } else {
                    throw new ResponseStatusException(
                            HttpStatus.CONFLICT,
                            "В корзине уже есть блюда из точки «" + existingCanteen.getName()
                                    + "». Оформите текущий заказ или очистите корзину."
                    );
                }
            }
        }

        Optional<CartItem> existingItemOpt = cartItemRepository.findByUserIdAndDishId(user.getId(), dish.getId());

        if (existingItemOpt.isPresent()) {
            CartItem existingItem = existingItemOpt.get();
            existingItem.setQuantity(existingItem.getQuantity() + request.getQuantity());
            if (request.getSpecialRequest() != null && !request.getSpecialRequest().isBlank()) {
                existingItem.setSpecialRequest(request.getSpecialRequest().trim());
            }
            cartItemRepository.save(existingItem);
        } else {
            CartItem newItem = cartMapper.toEntity(request, user, dish);
            cartItemRepository.save(newItem);
        }

        List<CartItem> updatedItems = cartItemRepository.findByUserIdOrderByAddedAtAsc(user.getId());
        return buildCartSummary(user.getId(), updatedItems);
    }

    @Transactional
    public CartSummaryDto updateItemQuantity(Long cartItemId, Integer quantity) {
        CartItem cartItem = cartItemRepository.findById(cartItemId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND, "Позиция корзины с ID " + cartItemId + " не найдена"
                ));

        Long userId = cartItem.getUser().getId();

        if (quantity == null || quantity <= 0) {
            cartItemRepository.delete(cartItem);
        } else {
            cartItem.setQuantity(quantity);
            cartItemRepository.save(cartItem);
        }

        List<CartItem> updatedItems = cartItemRepository.findByUserIdOrderByAddedAtAsc(userId);
        return buildCartSummary(userId, updatedItems);
    }

    @Transactional
    public CartSummaryDto removeCartItem(Long cartItemId) {
        CartItem cartItem = cartItemRepository.findById(cartItemId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND, "Позиция корзины с ID " + cartItemId + " не найдена"
                ));
        Long userId = cartItem.getUser().getId();
        cartItemRepository.delete(cartItem);

        List<CartItem> updatedItems = cartItemRepository.findByUserIdOrderByAddedAtAsc(userId);
        return buildCartSummary(userId, updatedItems);
    }

    @Transactional
    public CartSummaryDto clearCart(Long userId) {
        findUserOrThrow(userId);
        cartItemRepository.deleteByUserId(userId);
        return buildCartSummary(userId, Collections.emptyList());
    }

    private CartSummaryDto buildCartSummary(Long userId, List<CartItem> items) {
        if (items == null || items.isEmpty()) {
            return cartMapper.toEmptySummaryDto(userId, DEFAULT_PICKUP_SLOTS);
        }

        Canteen canteen = items.get(0).getDish().getCanteen();
        List<CartItemDto> itemDtos = cartMapper.toItemDtoList(items);

        int totalQty = 0;
        int availableItemsCount = 0;
        BigDecimal totalAmount = BigDecimal.ZERO;
        int totalCalories = 0;
        double totalProteins = 0.0;
        double totalFats = 0.0;
        double totalCarbs = 0.0;
        int maxPrepTime = 0;

        for (CartItem item : items) {
            Dish d = item.getDish();
            // Если блюдо в стоп-листе — оно отображается в списке со статусом «Закончилось»,
            // но НЕ учитывается в итоговой сумме, количестве и калорийности!
            if (d == null || !Boolean.TRUE.equals(d.getIsAvailable())) {
                continue;
            }

            int qty = item.getQuantity();
            availableItemsCount++;
            totalQty += qty;
            totalAmount = totalAmount.add(d.getPrice().multiply(BigDecimal.valueOf(qty)));
            totalCalories += d.getCalories() * qty;
            totalProteins += d.getProteins() * qty;
            totalFats += d.getFats() * qty;
            totalCarbs += d.getCarbs() * qty;

            if (d.getPrepTimeMinutes() != null && d.getPrepTimeMinutes() > maxPrepTime) {
                maxPrepTime = d.getPrepTimeMinutes();
            }
        }

        return cartMapper.toSummaryDto(
                userId,
                canteen,
                itemDtos,
                totalQty,
                availableItemsCount,
                totalAmount.setScale(2, RoundingMode.HALF_UP),
                totalCalories,
                roundMacro(totalProteins),
                roundMacro(totalFats),
                roundMacro(totalCarbs),
                maxPrepTime,
                DEFAULT_PICKUP_SLOTS
        );
    }

    private double roundMacro(double value) {
        return BigDecimal.valueOf(value).setScale(1, RoundingMode.HALF_UP).doubleValue();
    }

    private User findUserOrThrow(Long userId) {
        return userRepository.findById(userId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND, "Пользователь с ID " + userId + " не найден"
                ));
    }
}

        Set<DietaryTag> tags = new LinkedHashSet<>();
        if (codes != null) {
            for (String code : codes) {
                if (code == null || code.isBlank()) continue;
                dietaryTagRepository.findByCodeIgnoreCase(code.trim())
                        .ifPresent(tags::add);
            }
        }

    private String storeDishImage(MultipartFile file) {
        String originalFilename = file.getOriginalFilename();
        String extension = ".png";

        if (originalFilename != null && originalFilename.contains(".")) {
            extension = originalFilename.substring(originalFilename.lastIndexOf(".")).toLowerCase();
        }

        if (!List.of(".png", ".jpg", ".jpeg", ".webp", ".gif").contains(extension)) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Недопустимый формат файла. Разрешены: PNG, JPG, JPEG, WEBP"
            );
        }

        try {
            byte[] bytes = file.getBytes();

            Path uploadDir = Paths.get("uploads", "dishes").toAbsolutePath().normalize();
            Files.createDirectories(uploadDir);
            String uniqueFileName = "dish_" + UUID.randomUUID() + extension;
            Files.write(uploadDir.resolve(uniqueFileName), bytes);

            String mimeType = switch (extension) {
                case ".jpg", ".jpeg" -> "image/jpeg";
                case ".webp" -> "image/webp";
                case ".gif" -> "image/gif";
                default -> "image/png";
            };

            return "data:" + mimeType + ";base64," + Base64.getEncoder().encodeToString(bytes);
        } catch (IOException e) {
            throw new ResponseStatusException(
                    HttpStatus.INTERNAL_SERVER_ERROR, "Ошибка при сохранении изображения блюда"
            );
        }
    }

            order.addItem(orderItem);

                user,

                "Предзаказ #" + savedOrder.getOrderNumber() + " принят",

                        HttpStatus.NOT_FOUND, "Отзыв с ID " + reviewId + " не найден"
                ));

        if (replyText == null || replyText.isBlank()) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Текст ответа не может быть пустым");
        }

        review.setVendorReply(replyText.trim());
        return reviewMapper.toDto(reviewRepository.save(review));
    }

    @Transactional
    public void deleteReview(Long reviewId) {
        Review review = reviewRepository.findById(reviewId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND, "Отзыв с ID " + reviewId + " не найден"
                ));
        Dish dish = review.getDish();
        reviewRepository.delete(review);
        reviewRepository.flush();
        recalculateDishAndCanteenRatings(dish);
    }

    private void recalculateDishAndCanteenRatings(Dish dish) {
        List<Review> dishReviews = reviewRepository.findByDishIdOrderByCreatedAtDesc(dish.getId());
        if (dishReviews.isEmpty()) {
            dish.setRating(5.0);
        } else {
            double avgDish = dishReviews.stream()
                    .mapToInt(Review::getRating)
                    .average()
                    .orElse(5.0);
            dish.setRating(roundRating(avgDish));
        }
        dishRepository.save(dish);

        Canteen canteen = dish.getCanteen();
        List<Review> canteenReviews = reviewRepository.findByDishCanteenIdOrderByCreatedAtDesc(canteen.getId());
        if (canteenReviews.isEmpty()) {
            canteen.setRating(5.0);
            canteen.setReviewsCount(0);
        } else {
            double avgCanteen = canteenReviews.stream()
                    .mapToInt(Review::getRating)
                    .average()
                    .orElse(5.0);
            canteen.setRating(roundRating(avgCanteen));
            canteen.setReviewsCount(canteenReviews.size());
        }
        canteenRepository.save(canteen);
    }

    private double roundRating(double val) {
        return BigDecimal.valueOf(val).setScale(1, RoundingMode.HALF_UP).doubleValue();
    }

@Service
@RequiredArgsConstructor
public class VendorOrderService {

    private final FoodOrderRepository foodOrderRepository;
    private final NotificationRepository notificationRepository;
    private final OrderMapper orderMapper;
    private final NotificationMapper notificationMapper;

    @Transactional(readOnly = true)
    public List<OrderResponseDto> getCanteenQueue(Long canteenId, OrderStatus status) {
        List<FoodOrder> orders = (status != null)
                ? foodOrderRepository.findByCanteenIdAndStatusOrderByCreatedAtAsc(canteenId, status)
                : foodOrderRepository.findByCanteenIdOrderByCreatedAtDesc(canteenId);
        return orderMapper.toResponseDtoList(orders);
    }

    @Transactional
    public OrderResponseDto updateOrderStatus(Long orderId, UpdateOrderStatusRequestDto request) {
        FoodOrder order = foodOrderRepository.findById(orderId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND, "Заказ с ID " + orderId + " не найден"
                ));

        OrderStatus oldStatus = order.getStatus();
        OrderStatus newStatus = request.getStatus();

        if (oldStatus == newStatus) {
            return orderMapper.toResponseDto(order);
        }

        order.setStatus(newStatus);
        order.setUpdatedAt(LocalDateTime.now());
        FoodOrder savedOrder = foodOrderRepository.save(order);

        generateNotificationForStatusChange(savedOrder, newStatus);

        return orderMapper.toResponseDto(savedOrder);
    }

    private void generateNotificationForStatusChange(FoodOrder order, OrderStatus status) {
        String title;
        String message;

        switch (status) {
            case COOKING -> {
                title = "Заказ готовится";
                message = "В столовой «" + order.getCanteen().getName() + "» начали готовить ваш заказ #" + order.getOrderNumber() + ".";
            }
            case READY -> {
                title = "Заказ готов к выдаче";
                message = "Заказ #" + order.getOrderNumber() + " готов к получению в «" + order.getCanteen().getName() + "».";
            }
            case COMPLETED -> {
                title = "Заказ выдан";
                message = "Заказ #" + order.getOrderNumber() + " выдан.";
            }
            case CANCELLED -> {
                title = "Заказ отменён";
                message = "Заказ #" + order.getOrderNumber() + " был отменён.";
            }
            default -> {
                title = "Статус заказа обновлён";
                message = "Статус заказа #" + order.getOrderNumber() + ": " + status.getDisplayName();
            }
        }

        Notification notification = notificationMapper.toEntity(
                order.getUser(), order, title, message
        );
        notificationRepository.save(notification);
    }
